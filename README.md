# learn-postgresql
CREATE TABLE IF NOT EXISTS books (
	id BIGINT NOT NULL,
	name VARCHAR(255),
	receipent BIGINT
) PARTITION BY RANGE (id);

CREATE SEQUENCE IF NOT EXISTS books_seq
	START WITH 1
	INCREMENT BY 1
	MINVALUE 1
	MAXVALUE 9999999999
	CACHE 1000;
;

ALTER TABLE books
ALTER COLUMN id SET DEFAULT (TO_CHAR(NOW(), 'YYYYMMDD') || LPAD(NEXTVAL('books_seq')::TEXT, 10, '0'))::BIGINT;

CREATE TABLE books_before_202503 PARTITION OF books
FOR VALUES FROM (0) TO (202503010000000000);
CREATE TABLE books_default PARTITION OF books DEFAULT;


CREATE TABLE books_before_202504 PARTITION OF books
FOR VALUES FROM (202503010000000000) TO (202505010000000000)
PARTITION BY HASH (receipent);
CREATE TABLE books_202504_p0 PARTITION OF books_before_202504
FOR VALUES WITH (MODULUS 4, REMAINDER 0);
CREATE TABLE books_202504_p1 PARTITION OF books_before_202504
FOR VALUES WITH (MODULUS 4, REMAINDER 1);
CREATE TABLE books_202504_p2 PARTITION OF books_before_202504
FOR VALUES WITH (MODULUS 4, REMAINDER 2);
CREATE TABLE books_202504_p3 PARTITION OF books_before_202504
FOR VALUES WITH (MODULUS 4, REMAINDER 3);
